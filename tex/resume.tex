\newsavebox{\resumeentrydatebox}
% 短标题与日期同一行；长标题完整显示，日期另起一行右对齐。
\newcommand{\resumeentry}[2]{%
  \par
  \sbox{\resumeentrytitlebox}{\entrytitle{#1}}%
  \sbox{\resumeentrydatebox}{\daterange{#2}}%
  \noindent
  \ifdim\dimexpr\wd\resumeentrytitlebox+\wd\resumeentrydatebox+1em\relax>\linewidth
    \entrytitle{#1}\par\nopagebreak
    \noindent\hfill\usebox{\resumeentrydatebox}%
  \else
    \usebox{\resumeentrytitlebox}\hfill\usebox{\resumeentrydatebox}%
  \fi
  \par\nopagebreak
}
% 项目符号占一格，续行与正文对齐；不额外增加段落空行。
\newcommand{\resumeitem}[1]{%
  \par\noindent\hangindent=1em\hangafter=1
  \makebox[1em][l]{$\bullet$}#1\par
}

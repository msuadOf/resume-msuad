% 可调缩放范围；上限不超过历史 11pt 正文及 1.18 倍行距。
\providecommand{\resumeminfontsize}{10}
\providecommand{\resumemaxfontsize}{11}
\providecommand{\resumeminbaseline}{11.5}
\providecommand{\resumemaxbaseline}{16.04}
\newsavebox{\resumepagebox}
\ExplSyntaxOn
\fp_new:N \l_resume_font_fp
\fp_new:N \l_resume_baseline_fp
\int_new:N \l_resume_trial_int
\cs_new_protected:Npn \resume_fit_page:
  {
    \int_zero:N \l_resume_trial_int
    \bool_do_until:nn
      {
        \dim_compare_p:nNn
          {\box_ht:N \resumepagebox + \box_dp:N \resumepagebox}
          < {\textheight + 0.01pt}
        || \int_compare_p:nNn {\l_resume_trial_int} > {20}
      }
      {
        \fp_set:Nn \l_resume_font_fp
          {min(11,\resumemaxfontsize) -
           (min(11,\resumemaxfontsize)-\resumeminfontsize)*\l_resume_trial_int/20}
        \fp_set:Nn \l_resume_baseline_fp
          {min(16.04,\resumemaxbaseline) -
           (min(16.04,\resumemaxbaseline)-\resumeminbaseline)*\l_resume_trial_int/20}
        \setbox\resumepagebox=\vbox
          {
            \hsize=\textwidth
            \fontsize{\fp_to_decimal:N \l_resume_font_fp pt}
              {\fp_to_decimal:N \l_resume_baseline_fp pt}\selectfont
            \resumecontent\par
          }
        \int_incr:N \l_resume_trial_int
      }
    \dim_compare:nNnT
      {\box_ht:N \resumepagebox + \box_dp:N \resumepagebox} > {\textheight}
      {\PackageError{resume}{Content~exceeds~one~page~at~minimum~size}
        {Reduce~content~or~adjust~the~minimum~size.}}
    \typeout{Resume~fit:~font=\fp_to_decimal:N \l_resume_font_fp pt,~
      baseline=\fp_to_decimal:N \l_resume_baseline_fp pt}
    \vbox to \textheight {\unvbox\resumepagebox}
  }
\cs_new_eq:NN \resumefitpage \resume_fit_page:
\ExplSyntaxOff
